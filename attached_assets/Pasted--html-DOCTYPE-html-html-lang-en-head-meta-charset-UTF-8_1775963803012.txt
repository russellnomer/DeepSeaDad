        // Navigation
        function switchPage(page) {
            document.querySelectorAll('.page-section').forEach(s => s.classList.remove('active'));
            const target = document.getElementById(`page-${page}`);
            if (target) target.classList.add('active');
            
            document.querySelectorAll('.nav-link').forEach(link => {
                link.classList.toggle('active', link.getAttribute('data-page') === page);
            });
            
            window.location.hash = page;
        }
        
        // Mobile menu
        document.getElementById('mobile-menu-btn').addEventListener('click', () => {
            const menu = document.getElementById('mobile-menu');
            menu.classList.toggle('hidden');
        });
        
        // Hash routing
        window.addEventListener('hashchange', () => {
            const hash = window.location.hash.slice(1) || 'home';
            switchPage(hash);
        });

            "Why don’t fish play piano? Because you can’t tuna fish! 🎹🐟",
            "What do you call a fish with no eyes? Fsh. 🐟",
            "Why did the fisherman go to school? To improve his net worth! 🎣"
        ];
        
        function showDadJoke() {
            const modal = document.getElementById('joke-modal');
            const textEl = document.getElementById('joke-text');
            textEl.textContent = jokes[Math.floor(Math.random() * jokes.length)];
            modal.classList.add('active');
        }
        
        function hideJokeModal() {
            document.getElementById('joke-modal').classList.remove('active');
        }
        
        // Moon phase
        function getMoonPhase() {
            const date = new Date();
            const cycle = 29.530588853;
            const days = (date.getTime() / 86400000) % cycle;
            const phaseIndex = Math.floor((days / cycle) * 8);
            const phases = [
                { emoji: "🌑", name: "New Moon" },
                { emoji: "🌒", name: "Waxing Crescent" },
                { emoji: "🌓", name: "First Quarter" },
                { emoji: "🌔", name: "Waxing Gibbous" },
                { emoji: "🌕", name: "Full Moon" },
                { emoji: "🌖", name: "Waning Gibbous" },
                { emoji: "🌗", name: "Last Quarter" },
                { emoji: "🌘", name: "Waning Crescent" }
            ];
            return phases[phaseIndex];
        }
        
        // Main conditions engine
        let currentLat = null;
        let currentLon = null;
        let currentMode = 'fresh';
        
        async function getConditions(lat, lon, label = null) {
            const loading = document.getElementById('loading-state');
            const empty = document.getElementById('empty-state');
            const results = document.getElementById('results-state');
            
            empty.classList.add('hidden');
            loading.classList.remove('hidden');
            results.classList.add('hidden');
            
            try {
                const weatherUrl = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=temperature_2m,wind_speed,wind_direction,precipitation,cloud_cover,sunrise,sunset&timezone=auto`;
                const marineUrl = `https://marine-api.open-meteo.com/v1/marine?latitude=${lat}&longitude=${lon}&current=water_temperature,wave_height`;
                
                const [wRes, mRes] = await Promise.all([
                    fetch(weatherUrl),
                    fetch(marineUrl).catch(() => ({ json: () => ({ current: {} }) }))
                ]);
                
                const weather = await wRes.json();
                const marine = await mRes.json();
                
                const now = new Date();
                const tempF = Math.round(weather.current.temperature_2m * 1.8 + 32);
                const windKmh = weather.current.wind_speed || 0;
                const windMph = Math.round(windKmh * 0.621371);
                const moon = getMoonPhase();
                
                // Populate metrics
                document.getElementById('temp-display').textContent = `${tempF}°F`;
                document.getElementById('wind-display').textContent = `${windMph} mph`;
                document.getElementById('moon-display').innerHTML = moon.emoji;
                document.getElementById('moon-name').textContent = moon.name;
                
                const waterTempEl = document.getElementById('water-temp-display');
                if (marine.current.water_temperature !== undefined) {
                    const waterF = Math.round(marine.current.water_temperature * 1.8 + 32);
                    waterTempEl.textContent = `${waterF}°F`;
                } else {
                    waterTempEl.textContent = '—';
                    waterTempEl.parentElement.style.opacity = '0.4';
                }
                
                // Fishing Score
                let score = 7;
                if (tempF > 75) score -= 2;
                if (tempF < 45) score -= 1;
                if (windMph > 15) score -= 3;
                if (weather.current.precipitation > 0) score += 1;
                if (moon.name.includes('Full') || moon.name.includes('New')) score += 2;
                score = Math.max(3, Math.min(10, score));
                
                const scoreEl = document.getElementById('score-number');
                const fillEl = document.getElementById('score-fill');
                scoreEl.textContent = score;
                fillEl.style.width = `${score * 10}%`;
                if (score >= 8) fillEl.className = 'fishing-score-fill score-excellent';
                else if (score >= 6) fillEl.className = 'fishing-score-fill score-good';
                else if (score >= 4) fillEl.className = 'fishing-score-fill score-fair';
                else fillEl.className = 'fishing-score-fill score-poor';
                
                document.getElementById('score-explain').textContent = 
                    score >= 8 ? 'Excellent conditions — fish are feeding hard today.' :
                    score >= 6 ? 'Good day on the water. Go early or late.' :
                    score >= 4 ? 'Fair conditions. Focus on structure and patience.' :
                    'Tough day. Bundle up and keep it simple.';
                
                // Summary banner
                const summaryText = document.getElementById('summary-text');
                summaryText.textContent = 
                    score >= 8 ? 'Excellent bite today. Get out early.' :
                    score >= 6 ? 'Good conditions. Aim for dawn or dusk.' :
                    score >= 4 ? 'Fair day. Fish slow and patient.' :
                    'Tough conditions. Keep expectations steady, kiddo.';
                
                // Dad Says structured advice
                const outlook = document.getElementById('dad-outlook');
                const windowEl = document.getElementById('dad-window');
                const behavior = document.getElementById('dad-behavior');
                const tactic = document.getElementById('dad-tactic');
                const gear = document.getElementById('dad-gear');
                const encourage = document.getElementById('dad-encourage');
                
                outlook.innerHTML = `<div class="advice-label">Fishing outlook</div><p>${tempF > 75 ? 'Hot day — fish are deep and slow.' : windMph > 15 ? 'Windy — stick to protected banks.' : 'Solid conditions today.'}</p>`;
                windowEl.innerHTML = `<div class="advice-label">Best window today</div><p>Use sunrise/sunset: fish active around dawn &amp; dusk.</p>`;
                behavior.innerHTML = `<div class="advice-label">Likely fish behavior</div><p>${currentMode === 'fresh' ? 'Looking for structure and shade.' : 'Feeding near current breaks.'}</p>`;
                tactic.innerHTML = `<div class="advice-label">Tactical suggestion</div><p>Match the hatch or use natural drift.</p>`;
                gear.innerHTML = `<div class="advice-label">Clothing / gear note</div><p>Light layers if it’s warm, rain jacket if precip is coming.</p>`;
                encourage.innerHTML = `<div class="advice-label">Dad’s encouragement</div><p>I’m proud of you for checking conditions before you even left the house, kiddo.</p>`;
                
                // Show results
                loading.classList.add('hidden');
                results.classList.remove('hidden');
                document.getElementById('refresh-btn').classList.remove('hidden');
                
                // Save to localStorage
                currentLat = lat;
                currentLon = lon;
                localStorage.setItem('deepseadad_location', JSON.stringify({ lat, lon, label: label || 'Your dock', mode: currentMode }));
                
                document.getElementById('last-updated').textContent = `Last updated ${now.toLocaleTimeString([], {hour:'numeric', minute:'2-digit'})}`;
                
            } catch (err) {
                console.error(err);
                loading.classList.add('hidden');
                
                // Graceful failure state
                results.classList.remove('hidden');
                document.getElementById('refresh-btn').classList.remove('hidden');
                
                const outlook = document.getElementById('dad-outlook');
                const windowEl = document.getElementById('dad-window');
                const behavior = document.getElementById('dad-behavior');
                const tactic = document.getElementById('dad-tactic');
                const gear = document.getElementById('dad-gear');
                const encourage = document.getElementById('dad-encourage');
                
                outlook.innerHTML = `<div class="advice-label">Fishing outlook</div><p>Couldn’t pull live data right now, but based on the season I’d still give it a shot.</p>`;
                windowEl.innerHTML = `<div class="advice-label">Best window today</div><p>Early morning or late evening is usually safest.</p>`;
                behavior.innerHTML = `<div class="advice-label">Likely fish behavior</div><p>They’ll still bite if you’re patient.</p>`;
                tactic.innerHTML = `<div class="advice-label">Tactical suggestion</div><p>Use what you know works in your water.</p>`;
                gear.innerHTML = `<div class="advice-label">Clothing / gear note</div><p>Just dress for the weather you can see.</p>`;
                encourage.innerHTML = `<div class="advice-label">Dad’s encouragement</div><p>I’m proud of you for trying anyway, kiddo. We’ll get live data next time.</p>`;
                
                document.getElementById('summary-text').textContent = 'Couldn’t reach the dock — but the fish are still out there.';
                document.getElementById('score-number').textContent = '—';
                document.getElementById('score-fill').style.width = '40%';
                document.getElementById('score-explain').textContent = 'Offline mode — use your own eyes and experience today.';
            }
        }
        
        function refreshConditions() {
            if (currentLat && currentLon) {
                getConditions(currentLat, currentLon);
            }
        }
        
        // Geolocation
        function getLocation() {
            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(pos => {
                    getConditions(pos.coords.latitude, pos.coords.longitude);
                }, () => {
                    document.getElementById('manual-btn').click();
                });
            }
        }
        
        // Manual location
        async function manualLocation() {
            const input = prompt("Enter your town or ZIP code, kiddo:");
            if (!input) return;
            
            try {
                const geoRes = await fetch(`https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(input)}&count=1&language=en&format=json`);
                const geoData = await geoRes.json();
                
                if (geoData.results && geoData.results.length > 0) {
                    const loc = geoData.results[0];
                    getConditions(loc.latitude, loc.longitude, loc.name);
                } else {
                    alert("Couldn’t find that location. Try a bigger city nearby.");
                }
            } catch(e) {
                alert("Sorry, kiddo — couldn’t look that up right now.");
            }
        }
        
        // Mode toggle
        function setMode(mode) {
            currentMode = mode;
            document.querySelectorAll('.mode-btn').forEach(btn => {
                btn.classList.toggle('active', btn.dataset.mode === mode);
            });
            
            if (currentLat && currentLon) {
                getConditions(currentLat, currentLon);
            }
            
            localStorage.setItem('deepseadad_mode', mode);
        }
        
        // Auto-restore on load
        function restoreSavedData() {
            const saved = localStorage.getItem('deepseadad_location');
            const savedMode = localStorage.getItem('deepseadad_mode');
            
            if (savedMode) {
                currentMode = savedMode;
                setMode(savedMode);
            }
            
            if (saved) {
                const data = JSON.parse(saved);
                currentLat = data.lat;
                currentLon = data.lon;
                getConditions(data.lat, data.lon, data.label);
            }
        }
        
        // Event listeners
        window.onload = () => {
            document.querySelectorAll('a[data-page]').forEach(link => {
                link.addEventListener('click', e => {
                    e.preventDefault();
                    switchPage(link.getAttribute('data-page'));
                });
            });
            
            const initialPage = window.location.hash.slice(1) || 'home';
            switchPage(initialPage);
            
            document.getElementById('get-location-btn').addEventListener('click', getLocation);
            document.getElementById('manual-btn').addEventListener('click', manualLocation);
            
            document.getElementById('mode-fresh').addEventListener('click', () => setMode('fresh'));
            document.getElementById('mode-salt').addEventListener('click', () => setMode('salt'));
            
            restoreSavedData();
            
            console.log('%c✅ Deep Sea Dad is live and ready on the dock. Warm, useful, and waiting for you every morning. 🪝', 'color:#e07a3f; font-family:monospace');
        };
    </script>
